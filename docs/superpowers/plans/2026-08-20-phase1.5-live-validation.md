# AI Research Observatory Phase 1.5 Live Validation Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Make Evidence globally idempotent across run dates, render daily reports from first-discovery records, and validate the core source pool with deterministic simulations plus one live smoke test.

**Architecture:** Replace date-scoped full Evidence files with one append-only `evidence/ledger.jsonl`. Each line is an `EvidenceRecord` containing its first-discovery `run_date` and immutable Evidence; `read_date()` filters that single ledger, so no cross-file transaction is required. Existing runner and report interfaces remain stable.

**Tech Stack:** Python 3.11+, dataclasses, standard-library JSON/Path, pytest 8+, existing Requests/BeautifulSoup adapters.

## Global Constraints

- `--date` is a run grouping date, not a remote publication-date filter.
- `published_at`, first `collected_at`, and first-discovery `run_date` remain distinct.
- An `evidence_id` may occur only once in the global ledger.
- A repeated snapshot on a later date must not appear in that later date's report.
- Preserve source health semantics; unavailable never means “no updates”.
- Keep runtime artifacts and credentials untracked.
- Do not add SQLite, Signal, Claim, Direction, schedulers, X integration, or external write behavior.
- Work on the currently checked-out `main` branch; do not create another branch.

```python
# src/ai_observatory/ledger.py
from __future__ import annotations

import json
from dataclasses import dataclass
from datetime import date
from pathlib import Path

from .evidence import Evidence


class LedgerCorruptionError(ValueError):
    pass


@dataclass(frozen=True)
class EvidenceRecord:
    schema_version: int
    run_date: date
    evidence: Evidence

    def to_dict(self) -> dict:
        return {
            "schema_version": self.schema_version,
            "run_date": self.run_date.isoformat(),
            "evidence": self.evidence.to_dict(),
        }

    @classmethod
    def from_dict(cls, payload: dict) -> "EvidenceRecord":
        return cls(
            schema_version=int(payload["schema_version"]),
            run_date=date.fromisoformat(payload["run_date"]),
            evidence=Evidence.from_dict(payload["evidence"]),
        )


class EvidenceLedger:
    def __init__(self, root: Path):
        self.root = root

    def _path(self) -> Path:
        return self.root / "ledger.jsonl"

    def _read_records(self) -> tuple[EvidenceRecord, ...]:
        path = self._path()
        if not path.exists():
            return ()
        records: list[EvidenceRecord] = []
        for line_number, line in enumerate(path.read_text(encoding="utf-8").splitlines(), start=1):
            if not line.strip():
                continue
            try:
                records.append(EvidenceRecord.from_dict(json.loads(line)))
            except (KeyError, TypeError, ValueError, json.JSONDecodeError) as exc:
                raise LedgerCorruptionError(f"invalid evidence ledger line {line_number}") from exc
        return tuple(records)

    def read_all(self) -> tuple[Evidence, ...]:
        return tuple(record.evidence for record in self._read_records())

    def read_date(self, run_date: date) -> tuple[Evidence, ...]:
        return tuple(
            record.evidence for record in self._read_records()
            if record.run_date == run_date
        )

    def append(self, run_date: date, evidence: Evidence) -> bool:
        records = self._read_records()
        if evidence.evidence_id in {record.evidence.evidence_id for record in records}:
            return False
        path = self._path()
        path.parent.mkdir(parents=True, exist_ok=True)
        record = EvidenceRecord(schema_version=1, run_date=run_date, evidence=evidence)
        serialized = json.dumps(record.to_dict(), ensure_ascii=False, sort_keys=True)
        with path.open("a", encoding="utf-8", newline="\n") as handle:
            handle.write(serialized + "\n")
        return True
```
